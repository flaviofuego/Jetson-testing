# Patched CMakeLists for LightweightMR create_mesh: Open3D removed
# (replaced by miniply.h); only CGAL + Eigen needed.
cmake_minimum_required(VERSION 3.10)
project(create_mesh)
set(CMAKE_CXX_STANDARD 14)
if(NOT CMAKE_BUILD_TYPE)
    set(CMAKE_BUILD_TYPE Release)
endif()

find_package(CGAL REQUIRED)
find_package(Eigen3 REQUIRED NO_MODULE)

add_executable(create_mesh create_mesh.cc)
target_link_libraries(create_mesh CGAL::CGAL Eigen3::Eigen)
